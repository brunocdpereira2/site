from flask import Flask, render_template, request, jsonify from datetime import datetime, timedelta import json import os app = Flask(__name__) # Arquivo para persistência de dados DATA_FILE = 'family_data.json' def load_data(): """Carrega dados do arquivo JSON""" if os.path.exists(DATA_FILE): with open(DATA_FILE, 'r', encoding='utf-8') as f: return json.load(f) return { 'members': [], 'tasks': [], 'activity_log': [] } def save_data(data): """Salva dados no arquivo JSON""" with open(DATA_FILE, 'w', encoding='utf-8') as f: json.dump(data, f, ensure_ascii=False, indent=2) save_data(data) return jsonify(member), 201 return jsonify(data['members']) @app.route('/api/members/', methods=['PUT', 'DELETE']) def member_detail(member_id): """Atualiza ou deleta um membro""" data data['tasks'][i].update({ 'title': task_data['title'], 'memberId': task_data['memberId'], 'days': task_data['days'], 'points': task_data['points'] }) save_data(data) return jsonify(data['tasks'][i]) datetime.now().strftime('%Y-%m-%d') key = f"{day}-{today}" for task in data['tasks']: if task['id'] == task_id: # Encontra o membro member = next((m for m in data['members'] if m['id'] == task['memberId']), None) if not member: return jsonify({'error': 'Membro não encontrado'}), 404 # Toggle da conclusão if key in task['completed']: del task['completed'][key] member['points'] -= task['points'] activity_type = 'uncompleted' points = -task['points'] else: task['completed'][key] = True member['points'] += task['points'] activity_type = 'completed' points = task['points'] #
